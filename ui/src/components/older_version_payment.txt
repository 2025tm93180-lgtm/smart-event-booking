import React, { useState } from 'react';
import { money } from '../utils';
import {Booking, PaymentStatusEnum, User} from '../types';
import { usePaymentApi } from '../hooks/usePaymentApi';
import { useBookingApi } from '../hooks/useBookingApi';

interface PaymentProps {
  booking: Booking;
  user: User | null;
  onSuccess: (paymentId: string, newBookingId?: string) => void;
  onCancel: () => void;
  onError: (message: string) => void;
}

export const Payment: React.FC<PaymentProps> = ({ booking, user, onSuccess, onCancel, onError }) => {
  const [loading, setLoading] = useState(false);
  const [status, setStatus] = useState<{ message: string; type: 'success' | 'error' | 'info' | null }>({
    message: '',
    type: null
  });

  const { fetchConfig, createOrder, logPayment, verifyPayment } = usePaymentApi();
  const { createBooking } = useBookingApi();

  const showStatus = (message: string, type: 'success' | 'error' | 'info') => {
    setStatus({ message, type });
  };

  const initiatePayment = async () => {
    const amountInPaise = Math.round(booking.amount * 100);

    try {
      setLoading(true);
      let storedBookingId = sessionStorage.getItem('bookingId');
      let realBookingId = '';

      if (storedBookingId) {
        // call booking api to know the latest status of payment
        try {
          const bookingResp = await fetch(`http://localhost:9091/api/v1/bookings/${storedBookingId}`);
          if (bookingResp.ok) {
            const bookingDataJson = await bookingResp.json();
            if (bookingDataJson.status === 'CONFIRMED') {
              onSuccess(bookingDataJson.paymentId || '', storedBookingId);
              sessionStorage.removeItem('bookingId');
              return;
            }
            if (bookingDataJson.status === 'PENDING') {
              showStatus(`Resuming pending booking: ${storedBookingId}`, 'info');
              realBookingId = storedBookingId;
            }
          }
        } catch (fetchErr) {
          console.warn('Failed to fetch existing booking, will try creating a new one', fetchErr);
        }
      }

      if (!realBookingId) {
        // 1. Create a booking to get real bookingId from the server
        showStatus('Initiating booking with server...', 'info');
        const bookingData = await createBooking({
          seatNumbers: booking.seats,
          venue: booking.event.venue,
          username: user?.name ?? 'Guest User',
          mail: user?.email ?? '',
          phoneNumber: user?.phone ?? '',
          eventId: booking.event.id,
          eventTitle: booking.event.title,
          eventDate: booking.event.date,
          eventStartTime: booking.event.startTime,
          eventDuration: booking.event.duration,
          city: booking.event.city,
          amount: booking.amount,
          currency: 'INR',
          status: 'PENDING',
        });

        if (!bookingData || !bookingData.bookingId) {
          throw new Error('Failed to create booking on server');
        }

        realBookingId = bookingData.bookingId;
        showStatus(`Booking created: ${realBookingId}. Fetching Razorpay configuration...`, 'info');
      } else {
        showStatus(`Using existing booking: ${realBookingId}. Fetching Razorpay configuration...`, 'info');
      }


      // 2. Fetch Razorpay configuration
      let razorpayKeyId = 'rzp_test_TmUoRlZbl3o0Vo';
      const configData = await fetchConfig();
      if (configData && configData.keyId) {
        razorpayKeyId = configData.keyId;
      }

      let orderData;
      try {
        orderData = await createOrder(amountInPaise, realBookingId);
      } catch (err) {
        console.warn('Backend order creation failed, simulating order for demo...', err);
        // Simulation for demo if backend is not ready
        orderData = {
          order_id: 'order_sim_' + Math.random().toString(36).substr(2, 9),
          amount: amountInPaise,
          currency: 'INR'
        };
      }

      showStatus('Order created: ' + orderData.order_id + '. Opening checkout...', 'info');

      // 4. Open Razorpay Modal
      // @ts-ignore
      if (typeof window.Razorpay === 'undefined') {
        showStatus('Razorpay SDK not loaded. Simulating successful payment for demo.', 'error');

        // Even in simulation, we try to call the backend if requested
        await logPayment({
          bookingId: realBookingId,
          amount: booking.amount,
          currency: 'INR',
          status: 'FAILED' as PaymentStatusEnum,
          razorpayOrderId: null,
          razorpayPaymentId: null,
          razorpaySignature: null,
          failureReason: null,
          timestamps: new Date().toISOString()
        },`sim_${realBookingId}_${new Date().getTime()}`);

        setTimeout(() => {
          onSuccess('pay_sim_' + Math.random().toString(36).substr(2, 9), realBookingId);
        }, 1500);
        return;
      }

      // @ts-ignore
      const options = {
        key: razorpayKeyId || undefined,
        amount: orderData.amount,
        currency: orderData.currency || 'INR',
        name: 'Smart Event Tickets',
        description: 'Payment for order #' + orderData.order_id,
        order_id: orderData.order_id,
        handler: async function (response: any) {
          console.log("[Payment] Razorpay handler triggered with response:", response);
          showStatus('Verifying payment and updating backend...', 'info');

          const timestamp = new Date().toISOString();

          try {
            await logPayment({
              bookingId: realBookingId,
              amount: booking.amount,
              currency: orderData.currency || 'INR',
              status: 'CONFIRMED' as PaymentStatusEnum,
              razorpayOrderId: response.razorpay_order_id,
              razorpayPaymentId: response.razorpay_payment_id,
              razorpaySignature: response.razorpay_signature,
              failureReason: null,
              timestamps: timestamp
            }, `pay_${realBookingId}_${response.razorpay_payment_id}`);

            // Original verification flow
            console.log("[Payment] Initiating verification with backend...");
            try {
              const verifyResult = await verifyPayment({
                razorpay_order_id: response.razorpay_order_id,
                razorpay_payment_id: response.razorpay_payment_id,
                razorpay_signature: response.razorpay_signature
              });
              console.log("verification result:", verifyResult);
            } catch (verifyErr: any) {
              console.warn('Server verification failed, but continuing for demo:', verifyErr);
              showStatus('Payment successful (Demo Mode)! Payment ID: ' + response.razorpay_payment_id, 'success');
              onSuccess(response.razorpay_payment_id, realBookingId);
              return;
            }

            showStatus('Payment successful and verified! Payment ID: ' + response.razorpay_payment_id, 'success');
            onSuccess(response.razorpay_payment_id, realBookingId);
          } catch (verifyErr: any) {
            console.warn('Verification/Backend update failed, but continuing for demo if simulated...', verifyErr);
            if (orderData.order_id.startsWith('order_sim_')) {
               showStatus('Simulated payment successful! Payment ID: ' + response.razorpay_payment_id, 'success');
               onSuccess(response.razorpay_payment_id, realBookingId);
            } else {
               showStatus('Payment verification failed: ' + verifyErr.message, 'error');
               onError(verifyErr.message);
            }
          } finally {
            setLoading(false);
          }
        },
        modal: {
          ondismiss: async function () {
            showStatus('Payment cancelled by user.', 'error');

            // Log cancellation to backend
            await logPayment({
              bookingId: realBookingId,
              amount: booking.amount,
              currency: 'INR',
              status: 'CANCELLED' as PaymentStatusEnum,
              razorpayOrderId: null,
              razorpayPaymentId: null,
              razorpaySignature: null,
              failureReason: 'User dismissed modal',
              timestamps: new Date().toISOString()
            }, `cancel_${realBookingId}_${new Date().getTime()}`);

            setLoading(false);
            onCancel();
          }
        },
        theme: {
          color: '#3399cc'
        }
      };

      // @ts-ignore
      const rzp = new window.Razorpay(options);
      rzp.on('payment.failed', async function (response: any) {
        const failureReason = response.error.description || response.error.reason || 'Transaction failed';
        showStatus('Payment failed: ' + failureReason, 'error');

        // Log failure to backend
        await logPayment({
          // uuId: crypto.randomUUID(),
          bookingId: realBookingId,
          amount: booking.amount,
          currency: 'INR',
          status: 'FAILED' as PaymentStatusEnum,
          razorpayOrderId: response.error.metadata ? response.error.metadata.order_id : null,
          razorpayPaymentId: response.error.metadata ? response.error.metadata.payment_id : null,
          razorpaySignature: null,
          failureReason: failureReason,
          timestamps: new Date().toISOString()
        }, `fail_${realBookingId}_${response.error.metadata ? response.error.metadata.payment_id : new Date().getTime()}`);

        setLoading(false);
        onError(failureReason);
      });

      rzp.open();

    } catch (err: any) {
      showStatus('Error: ' + err.message, 'error');
      setLoading(false);
      onError(err.message);
    }
  };
  const seatsToDisplay = booking.seats.join(', ');

  return (
    <div className="payment-card">
      <h3>Complete your Payment</h3>

      <div className="payment-summary">
        <div className="payment-row">
          <span>Amount to Pay</span>
          <strong>{money(booking.amount)}</strong>
        </div>
        <div className="payment-row">
          <span>Booking ID</span>
          <code>{booking.id}</code>
        </div>
        <div className="payment-row">
          <span>Seats</span>
          <code>{seatsToDisplay}</code>
        </div>
      </div>

      <button
        id="pay-btn"
        className="button button-primary full-button"
        onClick={initiatePayment}
        disabled={loading}
      >
        {loading ? 'Processing...' : 'Pay Now with Razorpay'}
      </button>

      {status.type && (
        <div className={`status-box ${status.type}`}>
          {status.message}
        </div>
      )}

      <p className="fine-print centered">
        Secure payment powered by Razorpay.
      </p>
    </div>
  );
};
